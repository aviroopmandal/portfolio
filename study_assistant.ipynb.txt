        "conn.execute(\"\"\"\n",
        "CREATE TABLE IF NOT EXISTS conversations (\n",
        "    id INTEGER PRIMARY KEY AUTOINCREMENT,\n",
        "    question TEXT NOT NULL,\n",
        "    answer TEXT NOT NULL,\n",
        "    evaluation TEXT\n",
        ")\n",
        "\"\"\")\n",
        "conn.execute(\"\"\"\n",
        "CREATE TABLE IF NOT EXISTS lessons (\n",
        "    id INTEGER PRIMARY KEY AUTOINCREMENT,\n",
        "    lesson TEXT NOT NULL\n",
        ")\n",
        "\"\"\")\n",
        "conn.commit()\n",
        "\n",
        "def save_conversation(question, answer, evaluation):\n",
        "    conn.execute(\n",
        "        \"INSERT INTO conversations(question, answer, evaluation) VALUES (?, ?, ?)\",\n",
        "        (question, answer, evaluation)\n",
        "    )\n",
        "    conn.commit()\n",
        "\n",
        "def save_lesson(lesson):\n",
        "    if lesson and lesson.strip():\n",
        "        conn.execute(\"INSERT INTO lessons(lesson) VALUES (?)\", (lesson.strip(),))\n",
        "        conn.commit()\n",
        "\n",
        "def get_lessons(limit=4):\n",
        "    rows = conn.execute(\n",
        "        \"SELECT lesson FROM lessons ORDER BY id DESC LIMIT ?\", (limit,)\n",
        "    ).fetchall()\n",
        "    return [row[0] for row in rows[::-1]]\n",
        "\n",
        "def get_conversation_memory(limit=3):\n",
        "    return conn.execute(\n",
        "        \"SELECT question, answer FROM conversations ORDER BY id DESC LIMIT ?\", (limit,)\n",
        "    ).fetchall()[::-1]\n",
        "\n",
        "print(\"SQLite memory ready.\")"

        "def format_sources(docs):\n",
        "    if not docs:\n",
        "        return \"No relevant source chunks were retrieved.\"\n",
        "    blocks = []\n",
        "    for i, doc in enumerate(docs, 1):\n",
        "        source = doc.metadata.get(\"source\", \"unknown\")\n",
        "        page = doc.metadata.get(\"page\", \"?\")\n",
        "        chunk_id = doc.metadata.get(\"chunk_id\", \"?\")\n",
        "        blocks.append(\n",
        "            f\"[S{i}] source={source}, page={page}, chunk={chunk_id}\\n{doc.page_content}\"\n",
        "        )\n",
        "    return \"\\n\\n\".join(blocks)"
      ]

      ]
    },
    {
      "cell_type": "code",

      "source": [
        "from typing import TypedDict\n",
        "import re\n",
        "from langgraph.graph import StateGraph, START, END\n",
        "\n",
        "class ResearchState(TypedDict, total=False):\n",
        "    question: str\n",
        "    answer: str\n",
        "    evaluation: str\n",
        "    reflection: str\n",
        "    attempts: int\n",
        "    sources: list\n",
        "\n",
        "MAX_ATTEMPTS = 2\n",
        "\n",
        "def research_node(state):\n",
        "    answer, docs = research_agent(\n",
        "        state[\"question\"],\n",
        "        state.get(\"reflection\", \"\")\n",
        "    )\n",
        "    return {\n",
        "        \"answer\": answer,\n",
        "        \"sources\": docs,\n",
        "        \"attempts\": state.get(\"attempts\", 0) + 1\n",
        "    }\n",
        "\n",
        "def evaluation_node(state):\n",
        "    evaluation = evaluator_agent(\n",
        "        state[\"question\"], state[\"answer\"], state.get(\"sources\", [])\n",
        "    )\n",
        "    return {\"evaluation\": evaluation}\n",
        "\n",
        "def reflection_node(state):\n",
        "    lesson = reflection_agent(\n",
        "        state[\"question\"], state[\"answer\"], state[\"evaluation\"]\n",
        "    )\n",
        "    return {\"reflection\": lesson}\n",
        "\n",
        "def route_after_evaluation(state):\n",
        "    match = re.search(r\"VERDICT:\\s*(PASS|FAIL)\", state.get(\"evaluation\", \"\"), re.I)\n",
        "    verdict = match.group(1).upper() if match else \"FAIL\"\n",
        "    if verdict == \"PASS\" or state.get(\"attempts\", 0) >= MAX_ATTEMPTS:\n",
        "        return \"save\"\n",
        "    return \"reflect\"\n",
        "\n",
        "def save_node(state):\n",
        "    save_conversation(\n",
        "        state[\"question\"], state[\"answer\"], state.get(\"evaluation\", \"\")\n",
        "    )\n",
        "    return {}\n",
        "\n",
        "graph = StateGraph(ResearchState)\n",
        "graph.add_node(\"research\", research_node)\n",
        "graph.add_node(\"evaluate\", evaluation_node)\n",
        "graph.add_node(\"reflect\", reflection_node)\n",
        "graph.add_node(\"save\", save_node)\n",
        "graph.add_edge(START, \"research\")\n",
        "graph.add_edge(\"research\", \"evaluate\")\n",
        "graph.add_conditional_edges(\n",
        "    \"evaluate\", route_after_evaluation,\n",
        "    {\"reflect\": \"reflect\", \"save\": \"save\"}\n",
        ")\n",
        "graph.add_edge(\"reflect\", \"research\")\n",
        "graph.add_edge(\"save\", END)\n",
        "research_workflow = graph.compile()\n",
        "print(\"LangGraph workflow compiled.\")"
      ]
    },
