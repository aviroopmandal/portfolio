        "\n",
        "    if mode == \"Summarise\":\n",
        "\n",
        "        return chain.invoke({\n",
        "            \"audience\": \"a general reader\",\n",
        "            \"task\": \"Summarise the supplied material.\",\n",
        "            \"text\": text\n",
        "        })\n",
        "\n",
        "    elif mode == \"Explain\":\n",
        "\n",
        "        return chain.invoke({\n",
        "            \"audience\": \"a secondary school student\",\n",
        "            \"task\": \"Explain the supplied material simply.\",\n",
        "            \"text\": text\n",
        "        })\n",
        "\n",
        "    elif mode == \"Calculate\":\n",
        "\n",
        "        result = chat_agent.invoke(\n",
        "            {\n",
        "                \"messages\": [\n",
        "                    {\n",
        "                        \"role\": \"user\",\n",
        "                        \"content\": text\n",
        "                    }\n",
        "                ]\n",
        "            },\n",
        "            config=config\n",
        "        )\n",
        "\n",
        "        return result[\"messages\"][-1].content\n",
        "\n",
        "    else:\n",
        "        return \"Invalid mode.\""
