        "\n",
        "@inventory_server.tool()\n",
        "def get_inventory():\n",
        "    \"\"\"Retrieve all products and their stock levels.\"\"\"\n",
        "\n",
        "    cursor = inventory_db.execute(\n",
        "        \"SELECT * FROM products\"\n",
        "    )\n",
        "\n",
        "    return [\n",
        "        {\n",
        "            \"product_id\": row[0],\n",
        "            \"name\": row[1],\n",
        "            \"quantity\": row[2],\n",
        "            \"unit_price\": row[3],\n",
        "            \"reorder_level\": row[4]\n",
        "        }\n",
        "        for row in cursor.fetchall()\n",
        "    ]\n",
        "\n",
        "\n",
        "@inventory_server.tool()\n",
        "def check_stock(product_id: int):\n",
        "    \"\"\"Check the stock level of a particular product.\"\"\"\n",
        "\n",
        "    cursor = inventory_db.execute(\n",
        "        \"SELECT * FROM products WHERE product_id = ?\",\n",
        "        (product_id,)\n",
        "    )\n",
        "\n",
        "    row = cursor.fetchone()\n",
        "\n",
        "    if row is None:\n",
        "        return {\"error\": \"Product not found\"}\n",
        "\n",
        "    return {\n",
        "        \"product_id\": row[0],\n",
        "        \"name\": row[1],\n",
        "        \"quantity\": row[2],\n",
        "        \"unit_price\": row[3],\n",
        "        \"reorder_level\": row[4]\n",
        "    }\n",
        "\n",
        "\n",
        "@inventory_server.tool()\n",
        "def add_stock(product_id: int, quantity: int):\n",
        "    \"\"\"Add stock to an existing product.\"\"\"\n",
        "\n",
        "    if quantity <= 0:\n",
        "        return {\"error\": \"Quantity must be positive\"}\n",
        "\n",
        "    cursor = inventory_db.execute(\n",
        "        \"UPDATE products SET quantity = quantity + ? \"\n",
        "        \"WHERE product_id = ?\",\n",
        "        (quantity, product_id)\n",
        "    )\n",
        "\n",
        "    inventory_db.commit()\n",
        "\n",
        "    if cursor.rowcount == 0:\n",
        "        return {\"error\": \"Product not found\"}\n",
        "\n",
        "    return {\n",
        "        \"message\": \"Stock updated successfully\",\n",
        "        \"product_id\": product_id,\n",
        "        \"added_quantity\": quantity\n",
        "    }\n",
        "\n",
        "\n",
        "@inventory_server.tool()\n",
        "def get_low_stock():\n",
        "    \"\"\"Identify products requiring stock replenishment.\"\"\"\n",
        "\n",
        "    cursor = inventory_db.execute(\"\"\"\n",
        "        SELECT name, quantity, reorder_level\n",
        "        FROM products\n",
        "        WHERE quantity <= reorder_level\n",
        "    \"\"\")\n",

        "            \"reorder_level\": row[2]\n",

        "    \"\"\"Calculate total income, expenses and profit.\"\"\"\n",
        "\n",

        "        SELECT\n",
        "            SUM(CASE WHEN transaction_type = 'income'\n",

        "                THEN amount ELSE 0 END)\n",
        "        FROM transactions\n",
        "    \"\"\")\n",
        "\n",
        "    income, expenses = cursor.fetchone()\n",
        "\n",
        "    income = income or 0\n",
        "    expenses = expenses or 0\n",
        "\n",
        "    return {\n",
        "        \"total_income\": income,\n",
